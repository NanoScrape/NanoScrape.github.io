curl -X POST "https://api.apify.com/v2/acts/santamaria-automations~immowelt-de-scraper/run-sync-get-dataset-items?token=YOUR_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{}'
// npm install apify-client
import { ApifyClient } from 'apify-client';

const client = new ApifyClient({ token: 'YOUR_TOKEN' });
const run = await client.actor('santamaria-automations/immowelt-de-scraper').call({});
const { items } = await client.dataset(run.defaultDatasetId).listItems();
console.log(items);
